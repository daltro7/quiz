Como Conectar a Planilha Google (Sem Complicação)

Siga estes 4 passos simples para receber todos os cliques e leads direto em uma planilha:

  1. Crie uma nova planilha no Google Planilhas (sheets.new) exclusiva para este funil.
  2. (Opcional) Nem precisa criar colunas manualmente! O script cria o cabeçalho completo sozinho no primeiro acesso.
  3. No menu da planilha, clique em: Extensões > Apps Script.
  4. Apague qualquer código que estiver lá e cole o código completo abaixo (com suporte a gravação e sincronização do Dashboard):
// 1. LEITURA DOS DADOS (QuizForge ➔ Planilha com Visitante Único)
function doGet(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var lastRow = sheet.getLastRow();
    if (lastRow <= 1) {
      return ContentService.createTextOutput(JSON.stringify({
        status: 'success', totalStarts: 0, totalCompletions: 0, totalLeads: 0, stepViews: {}, recentLeads: []
      })).setMimeType(ContentService.MimeType.JSON);
    }
    var lastCol = sheet.getLastColumn();
    var rows = sheet.getRange(1, 1, lastRow, lastCol).getValues();
    var headers = rows[0], colMap = {};
    for (var c = 0; c < headers.length; c++) colMap[String(headers[c] || '').trim().toLowerCase()] = c;
    function getCol(row, candidates) {
      for (var j = 0; j < candidates.length; j++) {
        var idx = colMap[candidates[j].toLowerCase()];
        if (idx !== undefined && row[idx] !== undefined) return row[idx];
      }
      return '';
    }
    var stepVisitors = {}, uniqueStarts = {}, uniqueCompletions = {}, leadsMap = {}, recentLeads = [];
    for (var i = 1; i < rows.length; i++) {
      var row = rows[i];
      var rowDate = getCol(row, ['data/hora', 'data', 'timestamp']);
      var rawSession = String(getCol(row, ['sessão (id)', 'sessão', 'sessao', 'session', 'sessao (id)']) || '').trim();
      var event = String(getCol(row, ['evento', 'event']) || '').trim();
      var stepIdx = String(getCol(row, ['etapa', 'step']) || '').trim();
      var leadName = String(getCol(row, ['nome']) || '').trim();
      var leadPhone = String(getCol(row, ['whatsapp', 'telefone']) || '').trim();
      var leadEmail = String(getCol(row, ['email', 'e-mail']) || '').trim();
      var points = getCol(row, ['pontos / score', 'pontos', 'score']);
      var src = String(getCol(row, ['src', 'src (rastreamento)']) || '').trim();
      var utmSrc = String(getCol(row, ['utm_source', 'source']) || '').trim();
      var utmCmp = String(getCol(row, ['utm_campaign', 'campaign']) || '').trim();

      var sessionKey = rawSession;
      if (!sessionKey) {
        var timeCluster = rowDate ? Utilities.formatDate(new Date(rowDate), "GMT", "yyyy-MM-dd_HH") + '_' + Math.floor(new Date(rowDate).getMinutes() / 15) : ('leg_' + i);
        sessionKey = (leadPhone || leadEmail || 'visitante') + '_' + timeCluster;
      }

      if (event.indexOf('Quiz_Etapa_') === 0) {
        var numStep = parseInt(event.replace('Quiz_Etapa_', ''), 10);
        if (!isNaN(numStep) && numStep >= 1) {
          var k = 'step_' + (numStep - 1);
          if (!stepVisitors[k]) stepVisitors[k] = {};
          stepVisitors[k][sessionKey] = true;
          if (numStep === 1) uniqueStarts[sessionKey] = true;
        }
      } else if (event === 'Quiz_Iniciado') {
        if (!stepVisitors['step_0']) stepVisitors['step_0'] = {};
        stepVisitors['step_0'][sessionKey] = true;
        uniqueStarts[sessionKey] = true;
      } else if (stepIdx && !isNaN(parseInt(stepIdx, 10)) && event !== 'Quiz_Resposta') {
        var num = parseInt(stepIdx, 10);
        if (num >= 1) {
          var k2 = 'step_' + (num - 1);
          if (!stepVisitors[k2]) stepVisitors[k2] = {};
          stepVisitors[k2][sessionKey] = true;
          if (num === 1) uniqueStarts[sessionKey] = true;
        }
      }
      if (event === 'Quiz_Concluido' || event === 'Oferta_Visualizada') uniqueCompletions[sessionKey] = true;
      if (leadName || leadPhone || leadEmail) {
        var lKey = leadPhone || leadEmail || (leadName + '_' + sessionKey);
        if (!leadsMap[lKey]) {
          leadsMap[lKey] = true;
          var originTag = src ? ('SRC: ' + src) : [utmSrc, utmCmp].filter(Boolean).join(' / ');
          recentLeads.unshift({
            date: rowDate ? Utilities.formatDate(new Date(rowDate), Session.getScriptTimeZone() || "GMT-3", "dd/MM/yyyy HH:mm") : '',
            name: leadName || 'Anônimo', whatsapp: leadPhone || '', email: leadEmail || '', score: points || 0, src: src || '-', utm: originTag || '-'
          });
        }
      }
    }
    var stepViews = {};
    for (var sKey in stepVisitors) stepViews[sKey] = Object.keys(stepVisitors[sKey]).length;
    if (recentLeads.length > 50) recentLeads = recentLeads.slice(0, 50);
    return ContentService.createTextOutput(JSON.stringify({
      status: 'success', totalStarts: Object.keys(uniqueStarts).length, totalCompletions: Object.keys(uniqueCompletions).length,
      totalLeads: Object.keys(leadsMap).length, stepViews: stepViews, recentLeads: recentLeads, lastUpdated: new Date().toISOString()
    })).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

// 2. GRAVAÇÃO DOS DADOS (Quiz do Visitante ➔ Planilha com Mapeamento Dinâmico)
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var lastCol = sheet.getLastColumn();
    if (lastCol === 0) {
      sheet.appendRow(['Data/Hora', 'Sessão (ID)', 'Evento', 'Quiz', 'Etapa', 'Pergunta / Título', 'Resposta', 'Pontos / Score', 'Nome', 'WhatsApp', 'Email', 'SRC', 'UTM_Source', 'UTM_Campaign', 'UTM_Medium', 'UTM_Content']);
      sheet.getRange(1, 1, 1, 16).setFontWeight('bold').setBackground('#e2e8f0');
      sheet.setFrozenRows(1);
      lastCol = 16;
    } else {
      var curHeaders = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
      var hasSrc = curHeaders.some(function(h) { return String(h).toUpperCase().indexOf('SRC') !== -1; });
      var hasSess = curHeaders.some(function(h) { return String(h).toLowerCase().indexOf('sess') !== -1; });
      if (!hasSrc) { lastCol++; sheet.getRange(1, lastCol).setValue('SRC').setFontWeight('bold').setBackground('#e2e8f0'); }
      if (!hasSess) { lastCol++; sheet.getRange(1, lastCol).setValue('Sessão (ID)').setFontWeight('bold').setBackground('#e2e8f0'); }
    }
    var data = {};
    if (e.postData && e.postData.contents) {
      try { data = JSON.parse(e.postData.contents); } catch(p) { data = e.parameter || {}; }
    } else { data = e.parameter || {}; }
    var utms = data.utms || {};
    var src = data.src || utms.src || utms.sck || utms.SRC || utms.SCK || '';
    var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    var colMap = {};
    for (var c = 0; c < headers.length; c++) colMap[String(headers[c] || '').trim().toLowerCase()] = c;
    var newRow = new Array(headers.length);
    for (var k = 0; k < newRow.length; k++) newRow[k] = '';
    function setVal(candidates, val) {
      for (var j = 0; j < candidates.length; j++) {
        var idx = colMap[candidates[j].toLowerCase()];
        if (idx !== undefined) { newRow[idx] = val; return; }
      }
    }
    setVal(['data/hora', 'data', 'timestamp'], new Date());
    setVal(['sessão (id)', 'sessão', 'sessao', 'session', 'sessao (id)'], data.sessionId || '');
    setVal(['evento', 'event'], data.event || '');
    setVal(['quiz', 'quiztitle', 'título do quiz'], data.quizTitle || '');
    setVal(['etapa', 'step'], (data.data && data.data.stepIndex) || '');
    setVal(['pergunta / título', 'pergunta / titulo', 'pergunta', 'título'], (data.data && (data.data.stepTitle || data.data.question)) || '');
    setVal(['resposta', 'answer'], (data.data && data.data.answer) || '');
    setVal(['pontos / score', 'pontos', 'score'], (data.data && data.data.points) !== undefined ? data.data.points : (data.totalScore || ''));
    setVal(['nome', 'leadname'], (data.lead && data.lead.name) || '');
    setVal(['whatsapp', 'telefone', 'phone'], (data.lead && data.lead.whatsapp) || '');
    setVal(['email', 'e-mail'], (data.lead && data.lead.email) || '');
    setVal(['src', 'src (rastreamento)'], src);
    setVal(['utm_source', 'source'], data.utm_source || utms.utm_source || '');
    setVal(['utm_campaign', 'campaign'], data.utm_campaign || utms.utm_campaign || '');
    setVal(['utm_medium', 'medium'], data.utm_medium || utms.utm_medium || '');
    setVal(['utm_content', 'content'], data.utm_content || utms.utm_content || '');
    sheet.appendRow(newRow);
    return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

Como Colocar a Ferramenta no GitHub Pages

Você pode hospedar tanto esta ferramenta quanto os quizzes exportados no GitHub de forma 100% gratuita:

  1. Crie uma conta gratuita no GitHub.com.
  2. Crie um novo repositório (ex: meu-quiz-builder).
  3. Faça o upload dos arquivos desta pasta (index.html, app.js, template-engine.js, styles.css).
  4. No repositório, vá em Settings > Pages.
  5. Em Branch, selecione main e clique em Save.
  6. Em 1 minuto o GitHub gerará um link online seguro com HTTPS (ex: https://seu-usuario.github.io/meu-quiz-builder)!

Como Funciona este Construtor de Quiz?

1

O que é um Funil de Quiz?

É uma página onde o visitante responde 3 ou 4 perguntas rápidas e, no final, recebe um diagnóstico com um botão para comprar o seu produto exato.

2

Painel da Esquerda vs Celular da Direita

Na esquerda você digita os títulos, adiciona vídeos do YouTube e configura os botões. Na direita aparece o celular do seu cliente. Você pode clicar nos botões do celular para testar a experiência na hora!

3

Como usar o quiz pronto?

Quando terminar de editar, basta clicar no botão verde "Exportar HTML" no topo. Ele baixa um único arquivo (ex: meu-quiz.html) que você pode abrir ou colocar no ar na sua hospedagem.
